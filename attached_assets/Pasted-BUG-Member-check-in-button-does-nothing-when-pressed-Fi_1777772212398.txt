Full implementation:
const handleCheckIn = async () => {
  try {
    setLoading(true);
    const token = await AsyncStorage.getItem('token');
    
    console.log('Checking in with token:', token);
    
    const response = await axios.post(
      `${API_URL}/attendance/checkin`,
      {},
      { headers: { Authorization: `Bearer ${token}` }}
    );
    
    console.log('Checkin response:', response.data);
    
    const { streak, attendance, notification } = response.data;
    
    // Update streak display
    setStreak(streak);
    
    // Show success animation
    setCheckedIn(true);
    
    // Show streak animation with new number
    animateStreak(streak);
    
  } catch (error) {
    console.error('Checkin error:', error.response?.data);
    console.error('Checkin status:', error.response?.status);
    
    if (error.response?.status === 400) {
      // Already checked in today
      setAlreadyCheckedIn(true);
    } else {
      showToast('Check-in failed. Please try again.', 'error');
    }
  } finally {
    setLoading(false);
  }
};
